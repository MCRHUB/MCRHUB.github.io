    const canvas = document.getElementById('canvas');
    const ctx = canvas.getContext('2d');
    let width, height;
    let particles = [];
    const text = 'MCRHUB';
    const particleCount = 800;

    function resize() {
      width = canvas.width = window.innerWidth;
      height = canvas.height = window.innerHeight;
    }

    class Particle {
      constructor(x, y, targetX, targetY) {
        this.x = x;
        this.y = y;
        this.targetX = targetX;
        this.targetY = targetY;
        this.vx = (Math.random() - 0.5) * 4;
        this.vy = (Math.random() - 0.5) * 4;
        this.size = Math.random() * 2 + 1;
        this.color = `hsl(${Math.random() * 60 + 200}, 100%, 60%)`;
      }

      update() {
        const dx = this.targetX - this.x;
        const dy = this.targetY - this.y;
        this.vx += dx * 0.00005;
        this.vy += dy * 0.00005;
        this.vx *= 0.98;
        this.vy *= 0.98;
        this.x += this.vx;
        this.y += this.vy;
      }

      draw() {
        ctx.beginPath();
        ctx.arc(this.x, this.y, this.size, 0, Math.PI * 2);
        ctx.fillStyle = this.color;
        ctx.fill();
        ctx.shadowBlur = 10;
        ctx.shadowColor = this.color;
      }
    }

    function init() {
      resize();
      ctx.fillStyle = '#ffffff';
      ctx.font = 'bold 120px Arial';
      ctx.textAlign = 'center';
      ctx.textBaseline = 'middle';
      const textMetrics = ctx.measureText(text);
      const textWidth = textMetrics.width;
      const textX = width / 2;
      const textY = height / 2;

      const imageData = ctx.getImageData(
        textX - textWidth / 2,
        textY - 60,
        textWidth,
        120
      );

      particles = [];
      const step = Math.floor(Math.sqrt((textWidth * 120) / particleCount));
      for (let y = 0; y < imageData.height; y += step) {
        for (let x = 0; x < imageData.width; x += step) {
          const index = (y * imageData.width + x) * 4;
          const alpha = imageData.data[index + 3];
          if (alpha > 128) {
            const targetX = textX - textWidth / 2 + x;
            const targetY = textY - 60 + y;
            particles.push(
              new Particle(
                Math.random() * width,
                Math.random() * height,
                targetX,
                targetY
              )
            );
          }
        }
      }
    }

    function animate() {
      ctx.fillStyle = 'rgba(10, 10, 10, 0.15)';
      ctx.fillRect(0, 0, width, height);
      particles.forEach((p) => {
        p.update();
        p.draw();
      });
      requestAnimationFrame(animate);
    }

    window.addEventListener('resize', init);
    init();
    animate();
  </script>

1、MTL 简介

基本信息

中文全称：市场到线索流程

英文全称：Market to Lead（市场到线索）

华为体系对应简称：AR（Acquisition to Revenue，获客到营收），二者本质一致，AR更偏向业务视角，MTL更偏向端到端流程视角。

核心定义

MTL是企业客户生命周期的前端入口流程，是连接市场推广与销售转化的核心桥梁，覆盖从市场获客、多渠道线索归集、线索分层培育、线索质量筛选，到输出合格线索（MQL/SQL）并移交销售部门的完整链路，核心目标是高效获取高质量客户线索，降低获客成本，为销售转化提供稳定、优质的来源，是CRM系统中营销自动化（MA）模块的核心支撑流程。

核心流程

  1. 市场规划与投入：明确目标客户画像、营销渠道布局，制定营销预算，确定单渠道投入比例及单线索目标成本；

  2. 多渠道获客：通过官网、表单、公众号、短视频、投放广告（SEM/信息流）、线下展会、社群引流、内容营销（白皮书/案例）、企微裂变等全渠道，获取潜在客户信息；

  3. 线索归集与清洗：将全渠道获取的线索统一归集到CRM系统，完成线索去重、信息校验（过滤空号、无效信息），补充基础标签（行业、规模、来源渠道）；

  4. 线索分层与培育：基于线索标签（基础标签、行为标签、意向标签）进行分层，对弱意向线索通过营销自动化（MA）工具开展培育（定时推送行业资讯、产品资料、活动邀请，触发自动化互动旅程）；

  5. 线索质量认证：通过线索打分模型（结合互动行为、资料完整度、意向程度），筛选出符合目标客户画像、有明确需求意向的合格线索，分为MQL（营销合格线索）和SQL（销售合格线索）；

  6. 线索移交与协同：将SQL移交销售部门跟进，同步线索培育记录、互动历史，建立市场与销售的协同机制，跟踪线索后续转化情况，及时优化营销策略。

关键指标

  1. 获客类：总线索获取量、单渠道线索量、渠道线索占比、曝光量、点击率（CTR）、落地页跳出率、新增私域用户数；

  2. 线索质量类：线索有效率、线索去重率、线索档案完整率、MQL生成量、SQL生成量、线索转MQL转化率、MQL转SQL转化率；

  3. 培育运营类：线索培育覆盖率、内容触达量、内容打开率、线索互动率、沉睡线索占比、线索老化率；

  4. 成本与效率类：单线索获取成本（CPL）、单MQL成本、营销总预算、单渠道ROI、线索平均培育周期；

  5. 协同类：线索移交总量、线索退回率、销售首次响应及时率、线索跟进完成率。

与CRM的关联

MTL是CRM系统营销模块（MA营销自动化）的核心流程，CRM系统为MTL提供全链路支撑：

  1. 线索统一管理（线索池、去重、标签）；

  2. 自动化培育（旅程配置、多渠道触达）；

  3. 线索打分与认证（自定义打分模型）；

  4. 数据统计与分析（渠道效果、培育效果、转化数据）；

  5. 市场与销售协同（线索移交、跟进记录同步）。

常用延伸补充

  • MQL（Marketing Qualified Lead）：营销合格线索，由市场部认证，符合目标客户画像、有初步需求意向，可移交销售跟进的线索；

  • SQL（Sales Qualified Lead）：销售合格线索，由销售部复核，确认有明确商机潜力、符合跟进标准，可进入LTC流程的线索；

  • 线索打分模型：MTL的核心工具，自定义打分规则（如浏览产品页面+5分、下载方案+10分、咨询客服+15分），自动区分线索质量，提升销售人效；

  • 营销自动化（MA）：MTL流程的核心支撑工具，可实现线索培育、内容触达、旅程自动化，替代人工重复工作，提升培育效率。
